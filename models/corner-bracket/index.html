The actual SCPF Python source used by this model.

scpf/model.py
"""One shared, editable CornerBracket shape WITHOUT screw holes.

The standalone demonstration and wooden HUB75 assembly import this single
geometry definition. While the screw-hole operations are producing native
SolidCraft warnings, keep the bracket as a clean L-shaped body with the
reviewed inner fillet and outer/elephant-foot chamfers. Historical hole
experiments remain only in the standalone reference.py, not here.
"""
from scpf import Polygon2D, Edges, Sketch, component


@component
class CornerBracket:
    arm_width = 24.0
    arm_height = 12.0
    horizontal_arm_length = 60.0
    vertical_arm_length = 60.0

    outer_corner_chamfer = 6.0
    arm_end_chamfer = 3.0
    side_edge_chamfer = 2.0
    elephant_foot_relief = 0.6
    inside_corner_radius = 4.0

    def build(self):
        w = self.arm_height
        h = self.horizontal_arm_length
        v = self.vertical_arm_length
        end = self.arm_end_chamfer
        outer = self.outer_corner_chamfer

        # One neutral local model in the XZ plane (+Z up, +X right).
        #
        #       D--------------------------------C
        #      /                                  \
        #     E                                    B
        #     |                                    |
        #     |         I--------------------------A
        #     |         |
        #     |         |
        #     F         |
        #      \        |
        #       G-------H
        #
        # A o_hor_tip    B o_hor_bevel  C o_hor
        # D o_corner     E o_vert       F o_vert_bevel
        # G o_vert_end   H v_tip        I i_corner
        #
        # I is always the component origin. Assemblies place and rotate
        # occurrences of this geometry; this model never knows where it is used.
        points = [
            ("i_corner", (0, 0)),
            ("o_hor_tip", (h, 0)),
            ("o_hor_bevel", (h, w - end)),
            ("o_hor", (h - end, w)),
            ("o_corner", (-w + outer, w)),
            ("o_vert", (-w, w - outer)),
            ("o_vert_bevel", (-w, -v + end)),
            ("o_vert_end", (-w + end, -v)),
            ("v_tip", (0, -v)),
        ]

        with Sketch(name="Profile", plane="XZ") as sketch:
            Polygon2D(
                points,
                plane="XZ",
                world=True,
            )

        body = sketch.extrude(self.arm_width, sym=True)

        body = body.fillet(
            self.inside_corner_radius,
            edges=Edges.from_vertices("i_corner"),
        )
        # Apply the larger broad-face chamfer before the tiny tip relief.
        # SolidCraft 0.42 correctly rejects a 2 mm chamfer when the adjacent
        # face has already been reduced to 0.6 mm by the tip relief.
        body = body.chamfer(
            self.side_edge_chamfer,
            edges=Edges.from_profile("o_*", sides="both"),
        )
        body = body.chamfer(
            self.elephant_foot_relief,
            edges=Edges.from_vertices("o_hor_tip", "v_tip"),
        )
        body = body.chamfer(
            self.elephant_foot_relief,
            edges=Edges.from_profile("i_corner", sides="both"),
        )

        # No mounting holes in this version. The historical four-hole
        # geometry caused warnings in native SolidCraft when replayed
        # inside the larger wooden frame. Keep this bracket reliable while
        # the mounting approach is reviewed.

        return body

SCPC command source

Separate explicit-command implementation for comparison.

scpc/model.py
"""CornerBracket in native SolidCraft commands (no dependency on SCPF).

This is intentionally explicit: the command IDs, sketch profile and
edge-selection points are the native SolidCraft API. SCPC does not recreate
the bracket geometry in Python; it simply sends these commands.

Important: the edge positions below are for the **current 60/12/24 mm
reference bracket**. Unlike the parametrized Forge implementation, changing
a dimension here also requires reviewing the edge-selector positions.
"""
from scpc import Script

script = Script()
for name, value in (
    ("arm_width", 24),
    ("arm_height", 12),
    ("horizontal_arm_length", 60),
    ("vertical_arm_length", 60),
    ("arm_end_chamfer", 3),
    ("outer_corner_chamfer", 6),
    ("inside_corner_radius", 4),
    ("elephant_foot_relief", 0.6),
    ("side_edge_chamfer", 2),
):
    script.add("parameters.add", name=name, expression=str(value), unit="mm")

# This is the same XZ outline as the reviewed standalone printed bracket.
# Native sketch.line accepts points and (optional) stable corner labels.
script.add("sketch.create", plane="XZ", name="Profile")
script.add("sketch.line",
    points=[
        [0, 0],
        ["horizontal_arm_length", 0],
        ["horizontal_arm_length", "-(arm_height - (arm_end_chamfer))"],
        ["horizontal_arm_length - (arm_end_chamfer)", "-(arm_height)"],
        ["(-(arm_height)) + (outer_corner_chamfer)", "-(arm_height)"],
        ["-(arm_height)", "-(arm_height - (outer_corner_chamfer))"],
        ["-(arm_height)", "-((-(vertical_arm_length)) + (arm_end_chamfer))"],
        ["(-(arm_height)) + (arm_end_chamfer)", "-(-(vertical_arm_length))"],
        [0, "-(-(vertical_arm_length))"],
    ],
    closed=True,
    ids=["i_corner", "o_hor_tip", "o_hor_bevel", "o_hor",
         "o_corner", "o_vert", "o_vert_bevel", "o_vert_end", "v_tip"],
)
script.add("sketch.finish")
script.add("solid.extrude", distance="arm_width",
           direction="symmetric", body_name="CornerBracket")

# One concave inner fillet and three native chamfers; no screw holes.
script.add("solid.fillet", edges=[[0, 0, 0]],
           radius=4, body="CornerBracket")
# These positions select the two outside broad faces, including the
# mitred end edges. They are reference coordinates, not Python CSG.
outside_edges = [
    [60, -12, 4.5], [58.5, -12, 10.5], [25.5, -12, 12],
    [-9, -12, 9], [-12, -12, -25.5],
    [-10.5, -12, -58.5], [-4.5, -12, -60],
    [60, 12, 4.5], [58.5, 12, 10.5], [25.5, 12, 12],
    [-9, 12, 9], [-12, 12, -25.5],
    [-10.5, 12, -58.5], [-4.5, 12, -60],
]
script.add("solid.chamfer", edges=outside_edges,
           distance="side_edge_chamfer", body="CornerBracket")
script.add("solid.chamfer", edges=[[60, 0, 0], [0, 0, -60]],
           distance="elephant_foot_relief", body="CornerBracket")
script.add("solid.chamfer", edges=[[30, -12, 0], [30, 12, 0]],
           distance="elephant_foot_relief", body="CornerBracket")
